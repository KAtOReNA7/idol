function PhotoView({photo,label,kind='档案资料照',className='',live=true,retryable=photo?.table==='ratings',priority=false,fit}:{photo:Photo|null;label:string;kind?:string;className?:string;live?:boolean;retryable?:boolean;priority?:boolean;fit?:'contain'}){
 const [failed,setFailed]=useState(false);const [attempt,setAttempt]=useState(0);useEffect(()=>{setFailed(false);setAttempt(0);},[photo?.recordId,photo?.table,photo?.index,live]);
 const src=photo&&live?`/api/media?table=${photo.table}&record=${photo.recordId}&index=${photo.index}${attempt?'&retry='+attempt:''}`:'';
 return <div className={'photo '+className}>{src&&!failed?photo?.table==='ratings'?<a href={src} target="_blank" rel="noopener noreferrer" aria-label={'放大'+kind}><img src={src} style={fit?{objectFit:fit}:undefined} alt={label+' · '+kind} loading={priority?'eager':'lazy'} fetchPriority={priority?'high':'auto'} onError={()=>setFailed(true)}/></a>:<img src={src} style={fit?{objectFit:fit}:undefined} alt={label+' · '+kind} loading={priority?'eager':'lazy'} fetchPriority={priority?'high':'auto'} onError={()=>setFailed(true)}/>:<div className="photo-empty">{!(src&&failed&&retryable)&&<ImageIcon size={28}/>}<span>{photo?'图片暂时无法加载':kind.includes('MV')?'暂无 MV 截图':'暂无照片'}</span>{src&&failed&&retryable&&<button className="text-button photo-retry" aria-label={'重试加载'+label+'的图片'} onClick={event=>{event.stopPropagation();setAttempt(n=>n+1);setFailed(false);}}>重试图片</button>}</div>}<span className="image-label">{kind}</span></div>;
}
