function MemberWorkCard({item,member,data,onOpen}:{item:MemberWork;member:Member;data:Catalog;onOpen:()=>void}){
 const [failed,setFailed]=useState(false),[attempt,setAttempt]=useState(0),[expanded,setExpanded]=useState(false);const p=item.photo;
 useEffect(()=>{setFailed(false);setAttempt(0);setExpanded(false);},[p?.recordId,p?.index]);
 const src=p&&data.mode==='live'?`/api/media?table=ratings&record=${p.recordId}&index=${p.index}${attempt?'&retry='+attempt:''}`:'';
 const number=(n:number|null)=>n===null?'待评':n.toFixed(1),label=memberDisplayName(member)+' · '+item.work.song+' · 本 MV 个人造型';
 const closePhoto=useCallback(()=>setExpanded(false),[]);
 return <article data-browse-anchor={'work-'+item.work.id} className="member-work-wrap member-work-card">
  <div className="member-work-card__art">{src&&!failed?<button type="button" className="member-work-photo-button" aria-haspopup="dialog" aria-label={'放大'+label} onClick={()=>setExpanded(true)}><img src={src} alt={label} loading="lazy" onError={()=>setFailed(true)}/><span className="member-photo-expand">查看大图</span></button>:<span className="member-work-card__empty">{failed?'图片暂时打不开':'本 MV 个人造型图暂缺'}</span>}{failed&&<button className="member-work-retry text-button" aria-label={'重新加载'+member.name+'的'+item.work.song+'造型图'} onClick={()=>{setAttempt(x=>x+1);setFailed(false);}}>重试图片</button>}</div>
  <button data-member-work-trigger={item.work.id} className="member-work-card__body" onClick={onOpen}><small className="member-work-card__meta">{names(data,item.work.groupIds)} · {item.work.videoType||'MV'}{item.work.version?' · '+item.work.version:''}</small><h3 className="member-work-card__title">{item.work.song||item.work.name}</h3><span className="member-work-card__scores">{([['个人表现',item.performance],['个人造型',item.styling]] as const).map(([label,value])=><span className="member-work-card__score" key={label}><small className="member-work-card__score-label">{label}</small><strong className="member-work-card__score-value" data-unrated={value===null}>{number(value)}</strong></span>)}</span><span className="member-work-card__open">查看作品<ArrowUpRight size={16}/></span></button>
  {expanded&&src&&<PhotoLightbox src={src} label={label} onClose={closePhoto}/>}
 </article>;
}
